#include <iostream>
#include <cmath>

#define STATES 4

struct Tooth
{
    int location;
    int type;
    int num_caries;
};

struct Type
{
    int duration;
    int profit;
};

struct Appointment
{
    int number;
    int duration;
};

void fill_gene(int* gene, int number, int num_teeth);

void get_best_combination(
    int* gene,
    int number,
    int& best_gene,
    int& max_profit,
    struct Type* types,
    struct Tooth* teeth,
    int* appointment_durations,
    int num_teeth);

int main()
{
    struct Tooth teeth[] = {
        {1, 1, 3},
        {2, 1, 1},
        {3, 2, 0},
        {4, 3, 1},
        {5, 3, 2},
        {6, 4, 1},
        {7, 4, 3},
        {8, 4, 3}
    };
    struct Type types[] = {
        {5, 20},
        {7, 30},
        {10, 35},
        {12, 42}
    };

    int appointment_durations[] = {60, 45, 50};
    int num_teeth = sizeof(teeth) / sizeof(teeth[0]);
    int gene[] = {0, 0, 0, 0, 0, 0, 0, 0};
    int combinations = (int) pow(STATES, num_teeth);
    int max_profit = 0;
    int best_gene = 0;

    for (int i = 0; i < combinations; i++)
    {
        fill_gene(gene, i, num_teeth);
        get_best_combination(
            gene,
            i,
            best_gene,
            max_profit,
            types,
            teeth,
            appointment_durations,
            num_teeth);
    }

    for (int i = 0; i < num_teeth; i++) gene[i] = 0;
    fill_gene(gene, best_gene, num_teeth);

    // Se mantiene la salida en español tal como lo solicita el ejemplo del documento
    for (int i = 0; i < num_teeth; i++)
    {
        if (gene[i] == 1) std::cout << "Cita: 1 Diente " << i+1 << std::endl;
    }
    for (int i = 0; i < num_teeth; i++)
    {
        if (gene[i] == 2) std::cout << "Cita: 2 Diente " << i+1 << std::endl;
    }
    for (int i = 0; i < num_teeth; i++)
    {
        if (gene[i] == 3) std::cout << "Cita: 3 Diente " << i+1 << std::endl;
    }
    std::cout << "La ganancia maxima es "<< max_profit << std::endl;

    return 0;
}

void fill_gene(int* gene, int number, int num_teeth)
{
    for (int i = 0; i < num_teeth; i++) gene[i] = 0;
    int j = 0;
    while (number > 0)
    {
        gene[j] = number % STATES;
        number = number / STATES;
        ++j;
    }
}

void get_best_combination(
    int* gene,
    int number,
    int& best_gene,
    int& max_profit,
    struct Type* types,
    struct Tooth* teeth,
    int* appointment_durations,
    int num_teeth)
{
    int profit = 0;
    int t1 = 0, t2 = 0, t3 = 0;

    for (int i = 0; i < num_teeth; i++)
    {
        int type_index = teeth[i].type - 1;
        int duration = types[type_index].duration * teeth[i].num_caries;
        int tooth_profit = teeth[i].num_caries * types[type_index].profit;

        if (gene[i] == 1)
        {
            t1 += duration;
            profit += tooth_profit;
        }
        if (gene[i] == 2)
        {
            t2 += duration;
            profit += tooth_profit;
        }
        if (gene[i] == 3)
        {
            t3 += duration;
            profit += tooth_profit;
        }
    }

    if (t1 <= appointment_durations[0] and t2 <= appointment_durations[1] and t3 <= appointment_durations[2] and max_profit < profit)
    {
        max_profit = profit;
        best_gene = number;
    }
}
